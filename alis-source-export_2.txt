package za.co.alis.controller;

import com.google.gson.*;
import za.co.alis.dao.*;
import za.co.alis.model.*;
import za.co.alis.util.OllamaClient;

import jakarta.servlet.ServletException;
import jakarta.servlet.annotation.MultipartConfig;
import jakarta.servlet.http.*;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.sql.SQLException;
import java.util.List;

@MultipartConfig
public class DocumentUploadServlet extends HttpServlet {

    protected void doPost(HttpServletRequest req, HttpServletResponse resp) throws ServletException, IOException {
        HttpSession session = req.getSession(false);
        if (session == null || session.getAttribute("user") == null) {
            resp.sendRedirect("login.jsp");
            return;
        }
        User user = (User) session.getAttribute("user");

        Part filePart = req.getPart("document");
        String fileName = filePart.getSubmittedFileName();
        String content = new String(filePart.getInputStream().readAllBytes(), StandardCharsets.UTF_8);

        // Save document with status 'processing'
        Document doc = new Document();
        doc.setUserId(user.getUserId());
        doc.setFilename(fileName);
        doc.setContent(content);
        doc.setDocumentHash(OllamaClient.computeSHA256(content));
        doc.setStatus("processing");
        doc.setIngestionSource("manual");

        DocumentDAO docDAO = new DocumentDAO();
        int docId;
        try {
            docId = docDAO.save(doc);
            doc.setDocumentId(docId);
        } catch (SQLException e) {
            throw new ServletException("Failed to save document", e);
        }

        // Build law rules context
        StringBuilder rulesCtx = new StringBuilder();
        LawRuleDAO ruleDAO = new LawRuleDAO();
        List<LawRule> rules;
        try {
            rules = ruleDAO.findAll();
            for (LawRule r : rules) {
                rulesCtx.append("- Act ID ").append(r.getActId())
                        .append(", Keyword: ").append(r.getKeyword())
                        .append(", Requirements: ").append(r.getRequirements())
                        .append(", Risk: ").append(r.getRiskLevel())
                        .append(", Suggestion: ").append(r.getSuggestion()).append("\n");
            }
        } catch (SQLException e) {
            try { docDAO.updateStatus(docId, "error"); } catch (SQLException ignored) {}
            throw new ServletException("Failed to load law rules", e);
        }

        // Call Ollama
        OllamaClient ollama = new OllamaClient();
        String aiJson;
        try {
            aiJson = ollama.analyzeDocument(content, rulesCtx.toString());
        } catch (Exception e) {
            try { docDAO.updateStatus(docId, "error"); } catch (SQLException ignored) {}
            throw new ServletException("AI analysis failed", e);
        }

        // Parse JSON
        JsonObject result = JsonParser.parseString(aiJson).getAsJsonObject();
        JsonArray clausesArr = result.getAsJsonArray("clauses");

        // Save clauses and summary reports
        ClauseDAO clauseDAO = new ClauseDAO();
        SummaryReportDAO reportDAO = new SummaryReportDAO();
        LawRule defaultRule = rules.isEmpty() ? null : rules.get(0);

        try {
            for (JsonElement elem : clausesArr) {
                JsonObject cObj = elem.getAsJsonObject();
                Clause clause = new Clause();
                clause.setDocumentId(docId);
                clause.setClauseText(cObj.get("clause_text").getAsString());
                clause.setRiskLevel(cObj.get("risk_level").getAsString());
                int clauseId = clauseDAO.save(clause);

                if (defaultRule != null) {
                    SummaryReport report = new SummaryReport();
                    report.setDocumentId(docId);
                    report.setUserId(user.getUserId());
                    report.setRuleId(defaultRule.getRuleId());
                    report.setSimilarityScore(85.0);
                    report.setComplianceStatus(cObj.get("risk_level").getAsString().equals("Low") ? "Compliant" : "Non-compliant");
                    report.setRiskFlag(cObj.get("risk_level").getAsString());
                    report.setSuggestionText(cObj.get("suggestion").getAsString());
                    report.setReportText("AI analysis for clause: " + cObj.get("clause_text").getAsString());
                    reportDAO.save(report);
                }
            }
            docDAO.updateStatus(docId, "active");
        } catch (SQLException e) {
            try { docDAO.updateStatus(docId, "error"); } catch (SQLException ignored) {}
            throw new ServletException("Failed to save analysis results", e);
        }

        resp.sendRedirect("analysis?docId=" + docId);
    }
}

        try (Connection conn = DBConnection.getConnection();

                c.setClauseText(rs.getString("clause_text"));

import za.co.alis.model.*;

public class DocumentDAO {
    public int save(Document doc) throws SQLException {
        String sql = "INSERT INTO documents (user_id, filename, content, document_hash, status, ingestion_source) VALUES (?, ?, ?, ?, ?, ?) RETURNING document_id";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, doc.getUserId());
            stmt.setString(2, doc.getFilename());
            stmt.setString(3, doc.getContent());
            stmt.setString(4, doc.getDocumentHash());
            stmt.setString(5, doc.getStatus());
            stmt.setString(6, doc.getIngestionSource());
            ResultSet rs = stmt.executeQuery();
            if (rs.next()) return rs.getInt(1);
        }
        return -1;
    }

    public Document findById(int docId) throws SQLException {
        String sql = "SELECT * FROM documents WHERE document_id = ?";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setInt(1, docId);
            ResultSet rs = stmt.executeQuery();
            if (rs.next()) {
                Document d = new Document();
                d.setDocumentId(rs.getInt("document_id"));
                d.setUserId(rs.getInt("user_id"));
                d.setFilename(rs.getString("filename"));
                d.setContent(rs.getString("content"));
                d.setDocumentHash(rs.getString("document_hash"));
                d.setStatus(rs.getString("status"));
                d.setIngestionSource(rs.getString("ingestion_source"));
                d.setUploadedAt(rs.getTimestamp("uploaded_at"));
                return d;
            }
        }
        return null;
    }

    public List<Document> findByUser(String userId) throws SQLException {
        List<Document> list = new ArrayList<>();
        String sql = "SELECT * FROM documents WHERE user_id = ? ORDER BY uploaded_at DESC";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, userId);
            ResultSet rs = stmt.executeQuery();
            while (rs.next()) {
                Document d = new Document();
                d.setDocumentId(rs.getInt("document_id"));
                d.setUserId(rs.getInt("user_id"));
                d.setFilename(rs.getString("filename"));
                d.setStatus(rs.getString("status"));
                d.setUploadedAt(rs.getTimestamp("uploaded_at"));
                list.add(d);
            }
        }
        return list;
    }

    public void updateStatus(int docId, String status) throws SQLException {
        String sql = "UPDATE documents SET status = ? WHERE document_id = ?";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, status);
            stmt.setInt(2, docId);
            stmt.executeUpdate();
        }
    }

package za.co.alis.dao;

import za.co.alis.model.User;
import za.co.alis.util.PasswordUtil;
import java.sql.*;

public class UserDAO {
    public void create(User user) throws SQLException {
        String sql = "INSERT INTO users (full_name, email, password_hash, role) VALUES (?, ?, ?, ?)";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, user.getFullName());
            stmt.setString(2, user.getEmail());
            stmt.setString(3, user.getPasswordHash());
            stmt.setString(4, user.getRole());
            stmt.executeUpdate();
        }
    }

    public User findByEmail(String email) throws SQLException {
        String sql = "SELECT * FROM users WHERE email = ?";
        try (Connection conn = DBConnection.getConnection();
             PreparedStatement stmt = conn.prepareStatement(sql)) {
            stmt.setString(1, email);
            ResultSet rs = stmt.executeQuery();
            if (rs.next()) {
                User u = new User();
                ResultSetMetaData meta = rs.getMetaData();
                int columnCount = meta.getColumnCount();
                boolean hasUserId = false;
                for (int i = 1; i <= columnCount; i++) {
                    String colName = meta.getColumnLabel(i).toLowerCase();
                    if ("user_id".equals(colName) || "id".equals(colName) || "userid".equals(colName)) {
                       u.setUserId(rs.getInt("user_id"));
                        hasUserId = true;
                        break;
                    }
                }
                if (!hasUserId) {
                    u.setUserId(0);
                }
                u.setFullName(getStringIfExists(rs, meta, "full_name", "fullname", "name"));
                u.setEmail(getStringIfExists(rs, meta, "email"));
                u.setPasswordHash(getStringIfExists(rs, meta, "password_hash", "passwordhash", "password"));
                u.setRole(getStringIfExists(rs, meta, "role"));
                u.setCreatedAt(getTimestampIfExists(rs, meta, "created_at", "createdat", "created"));
                return u;
            }
        }
        return null;
    }

    private String getStringIfExists(ResultSet rs, ResultSetMetaData meta, String... candidates) throws SQLException {
        for (int i = 1; i <= meta.getColumnCount(); i++) {
            String colName = meta.getColumnLabel(i).toLowerCase();
            for (String candidate : candidates) {
                if (candidate.equals(colName)) {
                    return rs.getString(i);
                }
            }
        }
        return null;
    }

    private java.sql.Timestamp getTimestampIfExists(ResultSet rs, ResultSetMetaData meta, String... candidates) throws SQLException {
        for (int i = 1; i <= meta.getColumnCount(); i++) {
            String colName = meta.getColumnLabel(i).toLowerCase();
            for (String candidate : candidates) {
                if (candidate.equals(colName)) {
                    return rs.getTimestamp(i);
                }
            }
        }
        return null;
    }

    public boolean validateLogin(String email, String plainPassword) throws SQLException {
        User user = findByEmail(email);
        return user != null && PasswordUtil.checkPassword(plainPassword, user.getPasswordHash());
    }
}

package za.co.alis.util;

import com.google.gson.*;
import okhttp3.*;
import java.io.IOException;
import java.security.MessageDigest;
import java.security.NoSuchAlgorithmException;
import java.nio.charset.StandardCharsets;
import java.util.concurrent.TimeUnit;

public class OllamaClient {
    private static final String OLLAMA_URL = "http://localhost:11434/api/generate";
    private static final String MODEL = "llama3"; // or qwen2.5-coder
    private final OkHttpClient client;
    private final Gson gson;

    public OllamaClient() {
        this.client = new OkHttpClient.Builder()
                .connectTimeout(60, TimeUnit.SECONDS)
                .readTimeout(180, TimeUnit.SECONDS)
                .build();
        this.gson = new Gson();
    }

    public String analyzeDocument(String documentText, String lawRulesContext) throws IOException {
        String prompt = buildPrompt(documentText, lawRulesContext);
        
        JsonObject reqBody = new JsonObject();
        reqBody.addProperty("model", MODEL);
        reqBody.addProperty("prompt", prompt);
        reqBody.addProperty("stream", false);
        reqBody.addProperty("format", "json");

        Request request = new Request.Builder()
                .url(OLLAMA_URL)
                .post(RequestBody.create(reqBody.toString(), MediaType.parse("application/json")))
                .build();

        try (Response response = client.newCall(request).execute()) {
            if (!response.isSuccessful()) throw new IOException("Ollama error: " + response);
            JsonObject jsonResp = gson.fromJson(response.body().string(), JsonObject.class);
            return jsonResp.get("response").getAsString();
        }
    }

    private String buildPrompt(String documentText, String lawRulesContext) {
        return """
            You are a legal AI assistant specialized in South African contract law.
            Analyze the following document and return a JSON object with this exact structure:
            {
              "deal_type": "string",
              "sector": "string",
              "clauses": [
                {
                  "clause_text": "exact text from document",
                  "risk_level": "Low|Medium|High|Critical",
                  "suggestion": "improvement suggestion if any"
                }
              ]
            }

            Relevant South African legislation and compliance rules:
            %s

            Document text:
            %s

            Important: Output only valid JSON. No extra commentary.
            """.formatted(lawRulesContext, documentText);
    }

    public static String computeSHA256(String content) {
        try {
            MessageDigest md = MessageDigest.getInstance("SHA-256");
            byte[] hash = md.digest(content.getBytes(StandardCharsets.UTF_8));
            StringBuilder sb = new StringBuilder();
            for (byte b : hash) sb.append(String.format("%02x", b));
            return sb.toString();
        } catch (NoSuchAlgorithmException e) {
            throw new RuntimeException(e);
        }
    }
}
